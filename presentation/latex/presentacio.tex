\begin{frame}{Fonaments del mètode magnetotel·lúric}
\begin{columns}[T,onlytextwidth]
  \column{0.47\textwidth}
    \begin{caixa}[accent2]{Fonts naturals i règim difusiu}
      Tempestes elèctriques ($f > 1$\,Hz) i vent solar ($f < 1$\,Hz) indueixen corrents al subsòl.
      \[
        \frac{\omega\varepsilon}{\sigma} \ll 1
        \;\Longrightarrow\;
        \nabla^{2}\symbf{E} = \mu\sigma\,\frac{\partial \symbf{E}}{\partial t}
      \]
      \par
      {\scriptsize\color{muted} $\omega = 2\pi f$: freqüència angular; $\varepsilon$: permitivitat; $\sigma$: conductivitat; $\mu$: permeabilitat magnètica.}
    \end{caixa}
    \vspace{-0.1cm}
    {\small Els camps no es propaguen com ones: \textbf{difonen} i decauen amb la profunditat.}
  \column{0.50\textwidth}
    \begin{caixa}[accent]{Tensor d'impedància}
      \[
        \begin{pmatrix} E_x \\ E_y \end{pmatrix}
        =
        \begin{pmatrix} Z_{xx} & Z_{xy} \\ Z_{yx} & Z_{yy} \end{pmatrix}
        \begin{pmatrix} H_x \\ H_y \end{pmatrix}
      \]
      \par
      \begin{center}\footnotesize
      \begin{tabular}{@{}c@{\hspace{1.1em}}c@{\hspace{1.1em}}c@{}}
        $\begin{pmatrix} 0 & Z \\ -Z & 0 \end{pmatrix}$ &
        $\begin{pmatrix} 0 & Z_E \\ Z_H & 0 \end{pmatrix}$ &
        $\begin{pmatrix} Z_{xx} & Z_{xy} \\ Z_{yx} & Z_{yy} \end{pmatrix}$ \\[3pt]
        {\medium 1D} & {\medium 2D} & {\medium\color{accent}3D} \\
        {\scriptsize\color{muted}estratificat} & {\scriptsize\color{muted}eixos de l'\textit{strike}} & {\scriptsize\color{accent}cas general}
      \end{tabular}
      \end{center}
      \vspace{-0.15cm}
      {\scriptsize\color{muted} $E_i$, $H_i$: components horitzontals dels camps elèctric i magnètic a la superfície.}
    \end{caixa}
\end{columns}
\note{El mètode es basa en les fluctuacions naturals del camp electromagnètic: les tempestes generen senyal per sobre d'1 hertz i el vent solar per sota.
A les freqüències d'interès els corrents de desplaçament són negligibles davant dels de conducció, i les equacions de Maxwell es redueixen
a una equació de difusió: el camp no es propaga com una ona, sinó que decau amb la profunditat.
A la superfície es mesura la relació entre el camp elèctric i el magnètic, que és el tensor d'impedància.
La seva forma depèn de la dimensionalitat: en 1D la diagonal és nul·la, en 2D també si girem als eixos de l'strike,
i en 3D les quatre components són diferents de zero, que és el cas de Llucmajor.}
\end{frame}

\begin{frame}{Resistivitat aparent, fase i profunditat de penetració}
\begin{columns}[T,onlytextwidth]
  \column{0.46\textwidth}
    \begin{caixa}[accent]{Observables}
      \vspace{-0.15cm}
      \[
        \rho_{ij}(f) = \frac{1}{2\pi f \mu}\,\bigl|Z_{ij}(f)\bigr|^{2},
        \qquad
        \phi_{ij}(f) = \arg Z_{ij}(f)
      \]
      \par
      {\scriptsize\color{muted} $\rho_{ij}$: resistivitat aparent; $\phi_{ij}$: fase; $f$: freqüència.}
    \end{caixa}
    \vspace{-0.1cm}
    \begin{caixa}[accent2]{Skin depth}
      \vspace{-0.15cm}
      \[
        \delta = \sqrt{\frac{2}{\omega\mu_0\sigma}} \approx 503\,\sqrt{\rho\,T}\ \ \text{m}
      \]
      \par
      {\scriptsize\color{muted} $\delta$: profunditat on el camp decau un factor $1/e$; $\rho = 1/\sigma$ en $\Omega{\cdot}$m; $T = 1/f$: període en s.}
    \end{caixa}
    \vspace{-0.1cm}
    {\small Banda $1000$--$0{,}1$\,Hz: prou alta per resoldre l'aqüífer de Llucmajor.}
  \column{0.51\textwidth}
    \figura{figures/fig02_resistivitat_materials.png}
    \peu{Resistivitat dels materials de l'escorça (adaptat de Rosell, 2013). $\rho_a$ és una resposta integrada:
    la inversió reconstrueix la resistivitat real $\rho(x,y,z)$.}
\end{columns}
\note{De cada component del tensor en traiem la resistivitat aparent i la fase, que són les dades que ajustarem.
Cal remarcar que la resistivitat aparent no és la resistivitat real: és una mitjana de tot el volum que travessa el camp.
La profunditat de penetració, o skin depth, creix amb l'arrel del producte resistivitat per període: com més període, més profund.
Per això fem servir freqüències altes, de mil a una dècima d'hertz, perquè ens interessen els primers centenars de metres.
A la dreta es veu per què la resistivitat és útil: les argiles i l'aigua salina són conductores, i les calcàries seques, molt resistives.}
\end{frame}

  \column{0.37\textwidth}
    {\small\medium Criteris físics per descartar punts}
    \vspace{0.1cm}
    \begin{enumerate}\small
      \item \textbf{Linealitat i causalitat:} $Z(f)$ és holomorfa $\Rightarrow$ corbes suaus i contínues.
      \item \textbf{Kramers--Kronig:} la fase depèn del pendent de $\log\rho_a$ $\Rightarrow$ cap salt aïllat.
      \item \textbf{Difusió:} la resposta varia de manera gradual amb el període.
    \end{enumerate}
    \vspace{0.1cm}
    \begin{nota}[muted]
      \scriptsize Filtres automàtics (Hampel, LOWESS): insuficients quan el soroll és elevat.
    \end{nota}
\end{columns}
\note{Abans d'invertir cal depurar les dades, perquè l'algorisme no gasti esforç ajustant punts sense sentit físic.
Els criteris són tres: la impedància és una funció lineal i causal, per tant suau; per Kramers-Kronig la fase està lligada al pendent de la resistivitat aparent,
de manera que un salt en una sense l'altra no és físic; i el caràcter difusiu fa que tot variï gradualment.
Vaig provar d'automatitzar-ho amb filtres de Hampel i LOWESS, però quan el soroll és elevat contaminen la tendència. L'exemple és l'estació 26.}
\end{frame}

\begin{frame}{Inversió amb ModEM: gradients conjugats no lineals}
\vspace{-0.1cm}
\begin{nota}[accent]
  \normalsize\vspace{-0.3cm}
  \[
    \Phi(\symbf{m}) =
    \underbrace{\bigl(\symbf{d}-f(\symbf{m})\bigr)^{T} C_d^{-1}\bigl(\symbf{d}-f(\symbf{m})\bigr)}_{\text{desajust de les dades}}
    \;+\;
    \lambda\,\underbrace{\bigl(\symbf{m}-\symbf{m}_0\bigr)^{T} C_m^{-1}\bigl(\symbf{m}-\symbf{m}_0\bigr)}_{\text{regularització}}
  \]
  \par
  {\scriptsize\color{muted}
  $\symbf{d}$: dades observades; $f(\symbf{m})$: resposta del model (problema directe); $C_d$: covariància de les dades ($\sigma_i^2$);
  $\symbf{m}_0$: model inicial; $C_m$: covariància del model (suavitzat 0,2 en $x,y,z$, dues passades, mar fix); $\lambda$: pes de la regularització.}
\end{nota}
\vspace{0.1cm}
\begin{columns}[T,onlytextwidth]
  \column{0.315\textwidth}
    \begin{caixa}[muted]{Problema directe}
      Diferències finites de les equacions de Maxwell en règim quasi-estàtic.
    \end{caixa}
  \column{0.315\textwidth}
    \begin{caixa}[muted]{Optimització NLCG}
      Gradient pel mètode adjunt, sense construir la matriu jacobiana.
    \end{caixa}
  \column{0.33\textwidth}
    \begin{caixa}[accent2]{Mesura de l'ajust}
      \vspace{-0.12cm}
      \[
        \text{RMS} = \sqrt{\frac{1}{N}\sum_{i=1}^{N}\left(\frac{d_i^{\,\text{obs}}-d_i^{\,\text{pred}}}{\sigma_i}\right)^{2}}
      \]
      \par
      {\scriptsize\color{muted}$N$: nombre de dades; $\sigma_i$: error de cada dada. \textit{Error floor}: $\sigma_i \geq \mathrm{ef}\cdot\sqrt{|Z_{xy}Z_{yx}|}$.}
    \end{caixa}
\end{columns}
\note{La inversió minimitza aquesta funció de cost. El primer terme mesura el desajust entre les dades i la resposta del model, ponderat pels errors.
El segon és la regularització: penalitza allunyar-se del model inicial de manera poc suau, i lambda controla l'equilibri entre tots dos.
El problema directe es resol per diferències finites, i l'algorisme de gradients conjugats no lineals calcula el gradient pel mètode adjunt,
cosa que evita construir la jacobiana sencera. L'ajust es quantifica amb l'RMS normalitzat per l'error de cada dada;
l'error floor imposa un error mínim perquè les dades amb errors nominals molt petits no dominin la inversió.}
\end{frame}

Com que la skin depth va amb l'arrel del període, passar de 10 a 1 segon redueix la profunditat màxima un factor 3,16,
i això permet retallar la malla i abaratir molt la inversió, sense perdre resolució on ens interessa.}

      \item \textbf{Perfil 2}: màxima profunditat del confinament prop de la 51; franja resistiva superficial a la 67.
      \item \textbf{Perfil 3}: mateix gruix mínim de l'aqüitard (estació 63) i una \alert{discontinuïtat nova} ben definida.
    \end{itemize}
\end{columns}
\note{Comparant amb els perfils d'Arango, les zones resistives i conductores coincideixen en rang de resistivitat i en posició, però el model 3D té més resolució.
Al perfil 1 reproduïm l'aprimament de l'aqüitard sota l'estació 6 i a partir de la 4. Al perfil 2, la profunditat màxima del confinament prop de la 51.
I al perfil 3, a més del mateix gruix mínim, el nou model resol una discontinuïtat clarament definida que el model previ no mostrava.}
\end{frame}

\section{Conclusions}
% =====================================================================

\begin{frame}{Conclusions}
\begin{columns}[T,onlytextwidth]
  \column{0.64\textwidth}
    \begin{enumerate}\footnotesize
      \setlength{\itemsep}{0.5em}
      \item Inversió \textbf{3D formal} (ModEM-NLCG) de 40 estacions: metodologia objectiva i reproduïble, enfront de l'ajust per assaig i error.
      \item Depuració manual, error floor del 5\,\% i $T \leq 1$\,s: RMS final de \alert{\textbf{4,67}} sense descartar la informació 3D de les diagonals.
      \item L'RMS s'ha de llegir amb $m^2$ i $\lambda$: un RMS baix pot ser \textbf{cosmètic} (DM 2).
      \item Tres capes: aqüífer lliure resistiu ($\sim$150\,m), aqüitard conductor ($\sim$10\ohmm, 200--700\,m) i aqüífer confinat ($\sim$100\ohmm).
      \item \textbf{Discontinuïtats de l'aqüitard} coherents amb Arango (2005) i amb les dades tèrmiques (IGME, 2003).
    \end{enumerate}
  \column{0.33\textwidth}
    \begin{caixa}[rho2]{La pregunta inicial}
      \footnotesize El model reforça la hipòtesi que l'aigua calenta \textbf{ascendeix des de l'aqüífer confinat} a través de les discontinuïtats de l'aqüitard.
    \end{caixa}
    \vspace{0.1cm}
    \begin{caixa}[accent2]{Línies futures}
      \begin{itemize}\scriptsize
        \item Incorporar els SEV de l'IGME al model inicial.
        \item Incloure el tipper si la qualitat de les dades ho permet.
      \end{itemize}
    \end{caixa}
\end{columns}
\note{Per concloure: he aplicat una inversió 3D formal a les 40 estacions, amb una metodologia objectiva i reproduïble.
La depuració manual amb error floor i l'acotació de períodes han portat l'RMS final a 4,67 sense renunciar a la informació 3D.
He mostrat que l'RMS s'ha de llegir sempre juntament amb la norma del model i la regularització.
El model resol tres capes i, sobretot, discontinuïtats de l'aqüitard coherents amb el treball previ i amb les dades tèrmiques,
cosa que reforça la hipòtesi que l'aigua calenta puja des de l'aqüífer confinat. Com a línies futures, integrar els sondeigs elèctrics de l'IGME al model inicial.}
\end{frame}
